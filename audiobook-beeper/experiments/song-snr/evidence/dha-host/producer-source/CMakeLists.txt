cmake_minimum_required(VERSION 3.16)
project(DhaCodec VERSION 1.0.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(DHA_TOP_LEVEL ON)
else()
  set(DHA_TOP_LEVEL OFF)
endif()
option(DHA_BUILD_ENCODER "Include the offline beam encoder" ON)
option(DHA_BUILD_CLI "Build the WAV/DHAF command line tool" ${DHA_TOP_LEVEL})
option(DHA_BUILD_TESTS "Build codec tests" ${DHA_TOP_LEVEL})

add_library(dha_codec src/dha_common.c src/dha_decode.c)
add_library(dha::codec ALIAS dha_codec)
set_target_properties(dha_codec PROPERTIES EXPORT_NAME codec C_STANDARD 99 C_STANDARD_REQUIRED YES
  C_EXTENSIONS NO VERSION ${PROJECT_VERSION} SOVERSION 1)
target_include_directories(dha_codec PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(DHA_BUILD_ENCODER)
  target_sources(dha_codec PRIVATE src/dha_encode.c)
  target_compile_definitions(dha_codec PUBLIC DHA_HAS_ENCODER=1)
  if(NOT MSVC)
    target_link_libraries(dha_codec PRIVATE m)
  endif()
else()
  target_compile_definitions(dha_codec PUBLIC DHA_HAS_ENCODER=0)
endif()
if(BUILD_SHARED_LIBS)
  target_compile_definitions(dha_codec PRIVATE DHA_BUILDING PUBLIC DHA_SHARED)
endif()
if(MSVC)
  target_compile_options(dha_codec PRIVATE /W4)
else()
  target_compile_options(dha_codec PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(DHA_BUILD_CLI AND DHA_BUILD_ENCODER)
  add_executable(dha_tool tools/dha.c)
  set_target_properties(dha_tool PROPERTIES OUTPUT_NAME dha C_STANDARD 99 C_STANDARD_REQUIRED YES)
  target_link_libraries(dha_tool PRIVATE dha::codec)
  install(TARGETS dha_tool RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(DHA_BUILD_TESTS)
  enable_testing()
  add_executable(dha_decoder_test tests/test_decode_only.c)
  target_link_libraries(dha_decoder_test PRIVATE dha::codec)
  add_test(NAME dha-decoder-only COMMAND dha_decoder_test)
  if(DHA_BUILD_ENCODER)
    add_executable(dha_codec_test tests/test_codec.c)
    target_link_libraries(dha_codec_test PRIVATE dha::codec)
    add_executable(dha_trajectory_test tests/test_trajectory.c)
    target_link_libraries(dha_trajectory_test PRIVATE dha::codec)
    if(NOT MSVC)
      target_link_libraries(dha_codec_test PRIVATE m)
      target_link_libraries(dha_trajectory_test PRIVATE m)
    endif()
    add_test(NAME dha-codec COMMAND dha_codec_test)
    add_test(NAME dha-trajectory COMMAND dha_trajectory_test)
  endif()
  if(TARGET dha_tool)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME dha-files COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py --cli $<TARGET_FILE:dha_tool>
        --output ${CMAKE_CURRENT_BINARY_DIR}/file-tests)
    endif()
  endif()
endif()

install(TARGETS dha_codec EXPORT DhaCodecTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/dha_codec.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT DhaCodecTargets NAMESPACE dha:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DhaCodec)
configure_package_config_file(cmake/DhaCodecConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/DhaCodecConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DhaCodec)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/DhaCodecConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/DhaCodecConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/DhaCodecConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DhaCodec)
